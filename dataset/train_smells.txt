// Code smell example - Long method (God Method)
public class UserProcessor {
    public void processUserData(String userData, String rules, String config, Database db, Logger logger) {
        String[] parts = userData.split(",");
        String name = parts[0];
        String email = parts[1];
        String age = parts[2];
        String country = parts[3];
        String phone = parts[4];
        
        if (name != null && email != null && age != null) {
            if (name.length() > 2 && email.contains("@")) {
                int ageInt = Integer.parseInt(age);
                if (ageInt > 0 && ageInt < 150) {
                    if (country != null && phone != null && rules != null) {
                        String[] ruleArray = rules.split(";");
                        for (String rule : ruleArray) {
                            if (rule.contains("validate")) {
                                boolean valid = true;
                                if (config.contains("strict")) {
                                    valid = false;
                                }
                                if (valid) {
                                    db.save(name);
                                    db.save(email);
                                    db.save(age);
                                    db.save(country);
                                    db.save(phone);
                                    logger.log("User processed: " + name);
                                }
                            }
                        }
                    }
                }
            }
        }
    }
}
